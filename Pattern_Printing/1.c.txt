#include <stdio.h>

int main(){
    int n;
    scanf("%d",&n);
    //    *
    //   * *
    //  *****
    // *     *
    for(int i = 1; i < n; i++){
        printf(" ");
    }
    printf("*\n");
    int k = 1;
    for(int i = 2; i <= n; i++){
        for(int j = 0; j < n-i; j++){
            printf(" ");
        }
        printf("*");
        for(int j = 1; j <= k; j++){
            if(i== (n/2+1))
                printf("*");
            else    printf(" ");
        }
        k += 2;
        printf("*\n");
    }
    return 0;
}
